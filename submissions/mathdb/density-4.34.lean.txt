open Filter

namespace Set

noncomputable abbrev partialDensity {beta : Type*} [Preorder beta]
    [LocallyFiniteOrderBot beta] (S : Set beta) (A : Set beta := Set.univ)
    (b : beta) : Real :=
  ((Inter.inter (Inter.inter S A) (Iio b)).ncard : Real) /
    ((Inter.inter A (Iio b)).ncard : Real)

def HasDensity {beta : Type*} [Preorder beta] [LocallyFiniteOrderBot beta]
    (S : Set beta) (density : Real) (A : Set beta := Set.univ) : Prop :=
  Tendsto (fun b : beta => S.partialDensity A b) atTop (nhds density)

end Set
